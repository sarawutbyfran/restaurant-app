function addMenuToCartPrompt(menuItemId, menuName, basePrice, catName) { if(!activeTableId) { alert('กรุณาคลิกเลือกโต๊ะ ซุ้ม หรือคิวหน้าร้านก่อนเลือกรายการอาหารครับ!'); return; } let finalPrice = basePrice; let optionsStr = []; if (catName === 'อาหารตามสั่ง') { const eggDao = document.getElementById(`egg-${menuItemId}`); const eggJiao = document.getElementById(`eggj-${menuItemId}`); if (eggDao && eggDao.checked) { optionsStr.push('ไข่ดาว'); finalPrice += 10; eggDao.checked = false; } if (eggJiao && eggJiao.checked) { optionsStr.push('ไข่เจียว'); finalPrice += 15; eggJiao.checked = false; } } const noteInput = document.getElementById(`note-${menuItemId}`); const note = noteInput ? noteInput.value.trim() : ''; if (noteInput) noteInput.value = ''; // รวมชื่อตัวเลือกเสริมและหมายเหตุเข้าไปในชื่อหลัก let displayName = menuName; if (optionsStr.length > 0) { displayName += ` (+${optionsStr.join(', ')})`; } if (note) { displayName += ` [โน๊ต: ${note}]`; } const existing = currentCart.find(i => i.menu_item_id === menuItemId && i.name === displayName); if (existing) { existing.quantity += 1; } else { currentCart.push({ menu_item_id: menuItemId, name: displayName, // บันทึกชื่อที่รวมตัวเลือกเสริมแล้ว price: finalPrice, quantity: 1, notes: note || null, options: optionsStr.length > 0 ? optionsStr.join(', ') : null }); } renderSummaryBox(); } function updateCartQuantity(index, delta) { currentCart[index].quantity += delta; if (currentCart[index].quantity <= 0) { currentCart.splice(index, 1); } renderSummaryBox(); } async function confirmAndSubmitOrder() { if (!activeTableId || currentCart.length === 0) return; try { const res = await fetch('/api/orders', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ table_id: activeTableId, items: currentCart.map(i => ({ menu_item_id: i.menu_item_id, name: i.name, // ส่งชื่อที่ต่อท้ายด้วยตัวเลือกเสริมเข้าสู่ระบบ quantity: i.quantity, price: i.price, notes: i.notes, options: i.options })) }) }); if(res.ok) { currentCart = []; renderSummaryBox(); fetchKitchenOrders(); alert('ส่งออเดอร์เข้าครัวสำเร็จ!'); } else { alert('ส่งออเดอร์ไม่สำเร็จ'); } } catch(err) { alert('เกิดข้อผิดพลาดในการเชื่อมต่อ'); } }
